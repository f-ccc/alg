#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N=2e5+5;
std::string toManacherss(const std::string &s ){
    std::string res;res.reserve(2*s.size()+1);
    res="@#";
    for(int i=0;i<s.size();i++)res+=s[i],res+='#';
    res+='%';
    return res;
}
std::vector<int> Manacher(const std::string & s){
    auto t=toManacherss(s);
    std::vector<int>p(t.size());
    for(int i=1,r=0,c=0;i<t.size()-1;i++){
        p[i]=(r>i?std::min(p[2*c-i],r-i):1);
        while(t[i-p[i]]==t[i+p[i]])++p[i];
        if(i+p[i]>r)r=i+p[i],c=i;
    }
    return p;
}
void fc() {
    std::string s;std::cin>>s;
    int n=s.size();
    
    auto p=Manacher(s);
    int ans=0;

    std::vector<int>l(n+2),r(n+2);
    for(int i=1;i<p.size()-1;i++){
        if(p[i]<=1)continue;
        int start=(i-p[i])/2;
        int len=p[i]-1;
        int end=(start+len-1);
        l[end]=std::max(l[end],len);
        r[start]=std::max(r[start],len);
    }

    for(int i=n-2;i>=0;i--)l[i]=std::max(l[i],l[i+1]-2);
    for(int i=1;i<=n;i++)r[i]=std::max(r[i],r[i-1]-2);

    for(int i=0;i<n;i++){
        if(l[i]&&r[i+1])ans=std::max(l[i]+r[i+1],ans);
    }

    std::cout<<ans<<"\n";
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int t = 1;
	// std::cin>>t;
	while (t--) fc();
	return 0;
}
