#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N=2e5+5;
std::vector<int>g[N];
int dfn[N],low[N],tot;
int n,a,b;
int ans=N;

void tarjan(int u){
    low[u]=dfn[u]=++tot;
    for(int v:g[u]){
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
            if(low[v]>=dfn[u]&&u!=a&&u!=b&&dfn[b]>=dfn[v]){
                ans=std::min(ans,u);
            }
        }else{
            low[u]=std::min(low[u],dfn[v]);
        }
    }
}
void fc() {
    std::cin>>n;
    int u,v;std::cin>>u>>v;
    while(u&&v){
        g[u].push_back(v);
        g[v].push_back(u);
        std::cin>>u>>v;
    }

    std::cin>>a>>b;
    tarjan(a);

    if(ans==N){
        std::cout<<"No solution";
    }else{
        std::cout<<ans;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
