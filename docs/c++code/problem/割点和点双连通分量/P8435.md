#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 5e5 + 5;
std::vector<int> g[N], cut[N];
int dfn[N], low[N], tot;
int stk[N], top;
int n, m, cnt;

void tarjan(int u, bool root) {
    dfn[u] = low[u] = ++tot;
    stk[++top] = u;
    if (root && g[u].empty()) {
        cut[++cnt].push_back(u);
        return;
    }
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v,false);
            low[u] = std::min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                ++cnt;
                do {
                    cut[cnt].push_back(stk[top]);
                } while (stk[top--] != v);
                cut[cnt].push_back(u);
            }
        } else {
            low[u] = std::min(low[u], dfn[v]);
        }
    }
}
void fc() {
    std::cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        std::cin >> u >> v;
        if(u==v)continue;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    for (int i = 1; i <= n; i++) {
        if (!dfn[i]) {
            tarjan(i, true);
        }
    }

    std::cout << cnt << "\n";
    for (int i = 1; i <= cnt; i++) {
        std::cout << cut[i].size() << " ";
        for (int v : cut[i]) std::cout << v << " ";
        std::cout << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
