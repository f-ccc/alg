```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N = 1e5 + 5;
std::vector<int> g[N], adj[N << 1];
int dfn[N], low[N], timer;
int stk[N], top;
int n, m, tot;
ll ans[N << 1];
void add(int u, int v) {
    adj[u].push_back(v);
    adj[v].push_back(u);
}
void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk[++top] = u;
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                int sq = ++tot;
                add(sq, u);
                do {
                    add(sq, stk[top]);
                } while (stk[top--] != v);
            }
        } else {
            low[u] = std::min(low[u], dfn[v]);
        }
    }
}
int dfs(int u, int fa) {
    int sum = 0;
    for (int v : adj[u]) {
        if (v != fa) {
            int s = dfs(v, u);
            ans[u] += (ll)(n - s) * s;
            sum += s;
        }
    }
    if (u <= n) sum++;
    ans[u] += (ll)(n - sum) * sum + n - 1;
    return sum;
}
void fc() {
    std::cin >> n >> m;
    tot = n;

    for (int i = 0; i < m; i++) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    tarjan(1);

    dfs(1, 0);

    for (int i = 1; i <= n; i++) {
        std::cout << ans[i] << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
