1763#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N = 1e5 + 5;
std::vector<int> g[N], adj[N << 1];
int dfn[N], low[N], timer;
int stk[N], top;
int n, m, tot;
ll c[N << 1], siz[N << 1];
ll sum,total;
void add(int u, int v) {
    adj[u].push_back(v);
    adj[v].push_back(u);
}
void tarjan(int u) {
    total++;
    dfn[u] = low[u] = ++timer;
    stk[++top] = u;
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                int sq = ++tot;
                add(sq, u);
                do {
                    add(sq, stk[top]);
                } while (stk[top--] != v);
            }
        } else {
            low[u] = std::min(low[u], dfn[v]);
        }
    }
}
void dfs(int u, int f) {
    ll w=(u<=n?-1:adj[u].size());
    siz[u]+=(u<=n);
    for (int v : adj[u]) {
        if(v==f)continue;
        dfs(v,u);
        sum+=2LL*w*siz[u]*siz[v];
        siz[u]+=siz[v];
    }
    sum+=2LL*w*siz[u]*(total-siz[u]);
}
void fc() {
    std::cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    tot = n;

    for(int i=1;i<=n;i++){
        total=0;
        if(!dfn[i]){ 
            tarjan(i);
            dfs(i,0);
        }
    }

    std::cout << sum << "\n";
}
