#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N = 2e5 + 5, LOG = 20;
int dep[N << 1], up[N << 1][LOG];
std::vector<int> g[N], adj[N << 1];
int dfn[N], low[N], timer;
int stk[N], top;
int n, m, tot;
int cnt[N << 1];
std::pair<int,int> edge[N];
void add(int u, int v) {
    adj[u].push_back(v);
    adj[v].push_back(u);
}
void tarjan(int u) {
    low[u] = dfn[u] = ++timer;
    stk[++top] = u;
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                int sq = ++tot;
                do {
                    add(sq, stk[top]);
                } while (stk[top--] != v);
                add(sq, u);
            }
        } else {
            low[u] = std::min(low[u], dfn[v]);
        }
    }
}

void dfs(int u, int f) {
    dep[u] = dep[f] + 1;
    up[u][0] = f;
    for (int i = 1; i < LOG; i++) up[u][i] = up[up[u][i - 1]][i - 1];
    for (int v : adj[u])
        if (v != f) dfs(v, u);
}
int lca(int u, int v) {
    if (dep[u] < dep[v]) std::swap(u, v);
    int d = dep[u] - dep[v];
    for (int i = LOG - 1; i >= 0; i--)
        if ((d >> i) & 1) u = up[u][i];
    if (u == v) return u;
    for (int i = LOG - 1; i >= 0; i--)
        if (up[u][i] != up[v][i]) u = up[u][i], v = up[v][i];
    return up[u][0];
}
void dfs1(int u,int f){
    for(int v:adj[u]){
        if(v!=f){
            cnt[v]+=cnt[u];
            dfs1(v,u);
        }
    }
}
void rebuild() {
    for (int i = 0; i < m; i++) {
        int fa = up[edge[i].first][0];
        int fb = up[edge[i].second][0];
        if (fa == fb || up[fa][0] == edge[i].second) {
            cnt[fa]++;
        } else
            cnt[fb]++;
    }
    for(int i=n+1;i<=tot;i++)if(cnt[i]==1)cnt[i]=0;
    dfs1(1,0);
}
int get(int u,int v){
    int f=lca(u,v);
    return cnt[u]+cnt[v]-cnt[f]-cnt[up[f][0]];
}
void fc() {
    std::cin >> n >> m;
    tot = n;
    for (int i = 0; i < m; i++) {
        int u, v;
        std::cin >> u >> v;
        edge[i] = {u, v};
        g[u].push_back(v);
        g[v].push_back(u);
    }
    tarjan(1);
    dfs(1,0);
    rebuild();

    int q;std::cin>>q;
    while(q--){
        int u,v;std::cin>>u>>v;
        std::cout<<get(u,v)<<"\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
