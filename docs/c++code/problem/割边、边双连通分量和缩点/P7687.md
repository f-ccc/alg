#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=1e5+5;
constexpr int M=2e6+5;

int n,m,k,l;
struct Edge{
    int u,v;
}e[M>>1];
int head[N],to[M],nxt[M],tot=1;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
int dfn[N],low[N],timer;
bool is_bridge[M>>1];
int acnt[N],bcnt[N],cnt; 

void tarjan(int u,int e){
    dfn[u]=low[u]=++timer;
    for(int i=head[u];i;i=nxt[i]){
        if((i^1)==e)continue;
        int v=to[i];
        if(!dfn[v]){
            tarjan(v,i);
            low[u]=std::min(low[u],low[v]);
            if(low[v]>dfn[u]){
                if(acnt[v]==0||acnt[v]==k||bcnt[v]==0||bcnt[v]==l){
                    is_bridge[i>>1]=true;
                    cnt++;
                }
            }
            acnt[u]+=acnt[v];
            bcnt[u]+=bcnt[v];
        }else if(dfn[v]<dfn[u]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }
}
void fc() {
    std::cin>>n>>m>>k>>l;
    for(int i=0;i<k;i++){
        int x;std::cin>>x;acnt[x]=1;
    }
    for(int i=0;i<l;i++){
        int x;std::cin>>x;bcnt[x]=1;
    }

    for(int i=1;i<=m;i++){
        std::cin>>e[i].u>>e[i].v;
        add(e[i].u,e[i].v);
        add(e[i].v,e[i].u);
    }

    for(int i=1;i<=n;i++)if(!dfn[i])tarjan(i,0);

    std::cout<<cnt<<"\n";
    for(int i=1;i<=m;i++){
        if(is_bridge[i]){
            std::cout<<e[i].u<<" "<<e[i].v<<"\n";
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
