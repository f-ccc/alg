#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=5005;
constexpr int M=10005;

int n,m;
int head[N],to[M<<1],nxt[M<<1],tot=1;
int dfn[N],low[N],timer;
int bel[N],ebcc_cnt;
int stk[N],top;  
int deg[N];
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
void tarjan(int u,int e){
    dfn[u]=low[u]=++timer;
    stk[++top]=u;
    for(int i=head[u];i;i=nxt[i]){
        if((i^1)==e)continue;
        int v=to[i];
        if(!dfn[v]){
            tarjan(v,i);
            low[u]=std::min(low[u],low[v]); 
        }else if(dfn[v]<dfn[u]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        ++ebcc_cnt; 
        do{
            bel[stk[top]]=ebcc_cnt;
        }while(stk[top--]!=u);
    }
}
struct Edge{
    int u,v;
}e[N];
void fc() {
    std::cin>>n>>m;
    for(int i=0;i<m;i++){
        int u,v;std::cin>>u>>v;
        e[i].u=u;e[i].v=v;
        add(u,v);
        add(v,u);
    }

    tarjan(1,0);

    int cnt=0;
    for(int i=0;i<m;i++){
        int u=bel[e[i].u],v=bel[e[i].v];
        if(u==v)continue;
        ++deg[u],++deg[v];
    }
    for(int i=1;i<=ebcc_cnt;i++)if(deg[i]==1)cnt++;
    std::cout<<(cnt+1)/2;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
