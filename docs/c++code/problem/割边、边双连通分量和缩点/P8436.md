#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=5e5+5;
constexpr int M=2e6+5;

int n,m;
int head[N],to[M<<1],nxt[M<<1],tot=1;
void add(int u,int v){
    nxt[++tot]=head[u];
    to[tot]=v;
    head[u]=tot;
}
int dfn[N],low[N],timer;
int stk[N],top;

int ebcc_siz[N],ebcc_arr[N],ebcc_l[N],ebcc_r[N],idx,ebcc_cnt;

void tarjan(int u,int e){
    dfn[u]=low[u]=++timer;
    stk[++top]=u;
    for(int i=head[u];i;i=nxt[i]){
        if((i^1)==e)continue;
        int v=to[i];
        if(!dfn[v]){
            tarjan(v,i);
            low[u]=std::min(low[u],low[v]);
        }else if(dfn[v]<dfn[u]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }
    if(dfn[u]==low[u]){
        ++ebcc_cnt;
        ebcc_siz[ebcc_cnt]=0;
        ebcc_l[ebcc_cnt]=idx+1;
        int v;
        do{
            v=stk[top--];
            ebcc_siz[ebcc_cnt]++;
            ebcc_arr[++idx]=v;
        }while(v!=u);
        ebcc_r[ebcc_cnt]=idx;
    }
}
void fc() {
    std::cin>>n>>m;

    for(int i=1;i<=m;i++){
        int u,v;std::cin>>u>>v;
        add(u,v);
        add(v,u);
    }

    for(int i=1;i<=n;i++)if(!dfn[i])tarjan(i,0);

    std::cout<<ebcc_cnt<<"\n";
    for(int i=1;i<=ebcc_cnt;i++){
        std::cout<<ebcc_siz[i]<<" ";
        for(int j=ebcc_l[i];j<=ebcc_r[i];j++){
            std::cout<<ebcc_arr[j]<<" ";
        }
        std::cout<<"\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
