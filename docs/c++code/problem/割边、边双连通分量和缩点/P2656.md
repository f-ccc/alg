constexpr int N = 1e5 + 5;
constexpr int M = 2e5 + 5;

int head[N], to[M], nxt[M], tot = 1;
void add(int u, int v) {
    nxt[++tot] = head[u];
    head[u] = tot;
    to[tot] = v;
}
int dfn[N], low[N], timer;
bool is_bridge[M >> 1];

void tarjan(int u, int edge) {
    dfn[u] = low[u] = ++timer;
    for (int i = head[u]; i; i = nxt[i]) {
        if ((i ^ 1) == edge)
            continue;
        int v = to[i];
        if (!dfn[v]) {
            tarjan(v, i);
            low[u] = std::min(low[u], low[v]);
            if (low[v] > dfn[u]) {
                is_bridge[i >> 1] = true;
            }
        } else if (dfn[v] < dfn[u]) {
            low[u] = std::min(low[u], dfn[v]);
        }
    }
}
class Solution {
public:
    vector<vector<int>> criticalConnections(int n,vector<vector<int>>& connections) {
        tot = 1;
        timer = 0;
        std::fill(head, head + n, 0);
        std::fill(dfn, dfn + n, 0);
        std::fill(is_bridge + 1, is_bridge + connections.size() + 1, false);

        for (int i = 0; i < connections.size(); i++) {
            auto& e = connections[i];
            add(e[0], e[1]);
            add(e[1], e[0]);
        }

        for (int i = 0; i < n; i++) {
            if (!dfn[i])
                tarjan(i, 0);
        }

        std::vector<std::vector<int>> ans;
        for (int i = 1; i <= connections.size(); i++) {
            if (is_bridge[i])
                ans.push_back(connections[i - 1]);
        }
        return ans;
    }
};
