#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=100005;
constexpr int M=1000005;
struct edge{
    int u,v;
}e[M];
int head[N],to[M],nxt[M],tot;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
int dfn[N],low[N],timer;
int scc[N],scc_cnt;
int stk[N],top,c[N];
struct Node{
    int mn,mx;
}scc_c[M];
int n,m,ans;

void tarjan(int u){
    dfn[u]=low[u]=++timer;
    stk[++top]=u;

    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
        }else if(!scc[v]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        ++scc_cnt;
        int v;
        int mn=102;
        int mx=0;
        do{
            v=stk[top--];
            scc[v]=scc_cnt;
            mx=std::max(mx,c[v]);
            mn=std::min(mn,c[v]);
        }while(v!=u);
        scc_c[scc_cnt].mx=mx;
        scc_c[scc_cnt].mn=mn;
    }
}

int cmx[N];
bool cb[N];
std::pair<int,bool> dfs(int u){
    if(cmx[u]!=-1)return {cmx[u],cb[u]};
    int umx=0;
    bool ok=false;
    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        auto [vmx,isok]=dfs(v);
        if(isok)umx=std::max(umx,vmx),ok=true;
    }
    ok|=(scc[n]==u?true:false);
    if(ok)umx=std::max(scc_c[u].mx,umx);
    cb[u]=ok;
    cmx[u]=umx;
    if(ok)ans=std::max(ans,umx-scc_c[u].mn);
    return {cmx[u],cb[u]};
}
void fc() {
    std::cin>>n>>m;
    for(int i=1;i<=n;i++)std::cin>>c[i];
    for(int i=0;i<m;i++){
        std::cin>>e[i].u>>e[i].v;
        int x;std::cin>>x;
        add(e[i].u,e[i].v);
        if(x==2)add(e[i].v,e[i].u);
    }

    for(int i=1;i<=n;i++)if(!dfn[i])tarjan(i);
    for(int i=1;i<=scc_cnt;i++)head[i]=0;tot=0;

    for(int i=0;i<m;i++){
        int u=scc[e[i].u],v=scc[e[i].v];
        if(u==v)continue;
        add(u,v);
    }

    for(int i=1;i<=scc_cnt;i++)cmx[i]=-1;
    dfs(scc[1]);
    std::cout<<ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
