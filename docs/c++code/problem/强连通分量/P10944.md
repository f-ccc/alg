```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

constexpr int N = 1005;
constexpr int M = 6005;

int head[N], to[M], nxt[M], tot;
void add(int u, int v) {
    nxt[++tot] = head[u];
    head[u] = tot;
    to[tot] = v;
}
int dfn[N], low[N], timer;
int scc[N], scc_cnt;
std::vector<int> stk;
bool in_stk[N];
int in[N];
struct edge {
    int u, v;
} e[M];
int n, m;
void init() {
    for (int i = 1; i <= n; i++) {
        head[i] = dfn[i] = low[i] = scc[i] = in[i] = 0;
        in_stk[i] = false;
    }
    timer = tot = 0;
    stk.clear();
    scc_cnt=0;
}
void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk.push_back(u);
    in_stk[u] = true;

    for (int i = head[u]; i; i = nxt[i]) {
        int v = to[i];
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
        } else if (in_stk[v]) {
            low[u] = std::min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        ++scc_cnt;
        int v;
        do {
            v = stk.back();
            stk.pop_back();
            scc[v] = scc_cnt;
            in_stk[v] = false;
        } while (v != u);
    }
}
void fc() {
    std::cin >> n >> m;
    init();
    for (int i = 0; i < m; i++) {
        std::cin >> e[i].u >> e[i].v;
        add(e[i].u, e[i].v);
    }

    for (int i = 1; i <= n; i++)
        if (!dfn[i]) tarjan(i);
    for (int i = 1; i <= n; i++) head[i] = 0;
    tot=0;

    for (int i = 0; i < m; i++) {
        int u = scc[e[i].u];
        int v = scc[e[i].v];
        if (u == v) continue;
        in[v]++;
        add(u, v);
    }

    std::queue<int> q;
    for (int i = 1; i <= scc_cnt; i++)
        if (!in[i]) q.push(i);

    int cnt = 0;
    while (q.size()) {
        int u = q.front();
        q.pop();
        if(q.size()){
            std::cout<<"No\n";return;
        }
        cnt++;

        for (int i = head[u]; i; i = nxt[i]) {
            int v = to[i];
            if (--in[v] == 0) q.push(v);
        }
    }

    std::cout << (cnt == scc_cnt ? "Yes" : "No")<<"\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    std::cin >> t;
    while (t--) fc();
    return 0;
}
