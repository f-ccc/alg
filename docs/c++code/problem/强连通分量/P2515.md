#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=105;
int w[N],v[N],d[N];

int head[N],to[N],nxt[N],tot;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
};
int dfn[N],low[N],timer;
int scc[N],scc_cnt;
int stk[N],top;

int scc_w[N],scc_v[N],scc_c[N],in[N],id[N];

int n,m;

void tarjan(int u){
    int x;
    x=dfn[u]=low[u]=++timer;
    stk[++top]=u; 

    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
        }else if(!scc[v]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }
    if(low[u]==dfn[u]){
        ++scc_cnt;
        int t;
        int sw=0,sv=0;
        do{
            t=stk[top--];
            scc[t]=scc_cnt;
            sw+=w[t];
            sv+=v[t];
        }while(t!=u);
        scc_w[scc_cnt]=sw;
        scc_v[scc_cnt]=sv;
    } 
}

void dfs(int u){
    id[++timer]=u;
    scc_c[u]=1;
    for(int i=head[u];i;i=nxt[i]){
        dfs(to[i]);
        scc_c[u]+=scc_c[to[i]];
    }
}
int dp[N][505];
int f(){
    for(int i=scc_cnt+1;i>=2;i--){
        int u=id[i];
        for(int j=0;j<=m;j++){
            dp[i][j]=dp[i+scc_c[u]][j];
            if(j>=scc_w[u]){
                dp[i][j]=std::max(dp[i][j],scc_v[u]+dp[i+1][j-scc_w[u]]);
            }
        }
    }

    return dp[2][m];
}
void fc() {
    std::cin>>n>>m;
    for(int i=1;i<=n;i++)std::cin>>w[i];
    for(int i=1;i<=n;i++)std::cin>>v[i];
    for(int i=1;i<=n;i++)std::cin>>d[i],add(d[i],i);

    for(int i=0;i<=n;i++)if(!dfn[i])tarjan(i);

    memset(head,0,sizeof(head));
    tot=timer=0;

    for(int i=1;i<=n;i++)if(scc[d[i]]!=scc[i]){
        add(scc[d[i]],scc[i]);in[scc[i]]++;
    }
    for(int i=1;i<=scc_cnt;i++){
        if(i!=scc[0]&&!in[i])add(scc[0],i);
    }
    dfs(scc[0]);

    std::cout<<f();
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
