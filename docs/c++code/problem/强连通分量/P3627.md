#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=500005; 

struct edge{
    int u,v;
}e[N];

int head[N],to[N],nxt[N],tot;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
int dfn[N],low[N],timer;
int scc[N],scc_cnt;
int stk[N],top;
int n,m,s,p;
int c[N],scc_c[N];
bool is[N],scc_is[N];
int dp[N];
bool scc_ok[N];

void tarjan(int u){
    dfn[u]=low[u]=++timer;
    stk[++top]=u;
    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
        }else if(!scc[v]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }

    if(dfn[u]==low[u]){
        ++scc_cnt;
        int v;
        bool ok=false;
        do{
            v=stk[top--];
            scc[v]=scc_cnt;
            scc_c[scc_cnt]+=c[v];
            ok|=is[v];
        }while(v!=u);
        scc_is[scc_cnt]=ok;
    }
}
void work(){
    for(int i=1;i<=n;i++)if(!dfn[i])tarjan(i);
}

std::pair<int,int> dfs(int u){
    if(dp[u]!=-1)return {dp[u],scc_ok[u]};
    int mx=0; 
    bool ok=false;
    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        auto [s,h]=dfs(v);
        if(h){
            mx=std::max(mx,s);
            ok=true;
        }
    }
    if(!ok)dp[u]=0;
    else dp[u]=mx;
    ok|=scc_is[u];
    if(ok)dp[u]+=scc_c[u];
    scc_ok[u]=ok;
    return {dp[u],scc_ok[u]};
}
void fc() {
    std::cin>>n>>m;
    for(int i=0;i<m;i++){
        std::cin>>e[i].u>>e[i].v;
        add(e[i].u,e[i].v);
    }
    for(int i=1;i<=n;i++){
        std::cin>>c[i];
    }
    std::cin>>s>>p;
    for(int i=0;i<p;i++){
        int x;std::cin>>x;
        is[x]=true;
    }
    work();

    for(int i=1;i<=scc_cnt;i++)head[i]=0;tot=0;

    for(int i=0;i<m;i++){
        int u=scc[e[i].u],v=scc[e[i].v];
        if(u!=v){
            add(u,v);
        }
    }

    for(int i=1;i<=scc_cnt;i++)dp[i]=-1;

    std::cout<<dfs(scc[s]).first;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
