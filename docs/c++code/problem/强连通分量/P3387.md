#include <bits/stdc++.h>
using namespace std;
using ll = long long;

constexpr int N=10005;
constexpr int M=100005;

int head[N],to[M],nxt[M],tot;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
int dfn[N],low[N],timer;
int scc[N],scc_cnt;
int stk[N],top;
int n,m,c[N],scc_c[N];
int in[N],sum[N];

void tarjan(int u){
    dfn[u]=low[u]=++timer;
    stk[++top]=u;

    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
        }else if(!scc[v]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }

    if(dfn[u]==low[u]){
        ++scc_cnt;
        int v;
        int sum=0;
        do{
            v=stk[top--];
            scc[v]=scc_cnt;
            sum+=c[v];
        }while(v!=u);
        scc_c[scc_cnt]=sum;
    }
}
struct edge{int u,v;}e[M];
std::set<std::pair<int,int>>s;

void fc() {
    std::cin>>n>>m;
    for(int i=1;i<=n;i++)std::cin>>c[i];
    for(int i=0;i<m;i++){
        std::cin>>e[i].u>>e[i].v;
        add(e[i].u,e[i].v);
    }

    for(int i=1;i<=n;i++)if(!dfn[i])tarjan(i);
    for(int i=1;i<=scc_cnt;i++)head[i]=0;
    tot=0;
    
    for(int i=0;i<m;i++){
        int u=scc[e[i].u],v=scc[e[i].v];
        if(u==v)continue;
        if(s.count({u,v}))continue;
        add(u,v);
        in[v]++;
        s.insert({u,v});
    }

    int ans=0;
    std::queue<int>q;
    for(int i=1;i<=scc_cnt;i++)if(in[i]==0){
        q.push(i);
        sum[i]=scc_c[i];
        ans=std::max(sum[i],ans);
    }

    while(q.size()){
        int u=q.front();q.pop();
        for(int i=head[u];i;i=nxt[i]){
            int v=to[i];
            if(--in[v]==0)q.push(v);
            sum[v]=std::max(sum[u]+scc_c[v],sum[v]);
            ans=std::max(ans,sum[v]);
        }
    }

    std::cout<<ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
