/*
  P1726 上白泽慧音
*/

#include <bits/stdc++.h>
using namespace std;
using ll = long long;

constexpr int N = 5005;
std::vector<int> g[N];
int dfn[N], low[N], timer = 0;
int scc[N], scc_cnt;
std::vector<int> stk;
bool in_stk[N];

int n, m;
std::vector<int> ans;
int mx, mn = N;

void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk.push_back(u);
    in_stk[u] = true;

    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
        } else if (in_stk[v]) {
            low[u] = std::min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        ++scc_cnt;
        int v;
        std::vector<int> s;
        int vmn = N;
        do {
            v = stk.back();
            stk.pop_back();
            in_stk[v] = false;
            scc[v] = scc_cnt;
            s.push_back(v);
            vmn = std::min(vmn, v);
        } while (v != u);
        if (s.size() > mx) {
            mx = s.size();
            mn=vmn;
            ans = std::move(s);
        } else if (s.size() == mx && mn > vmn) {
            mn = vmn;
            ans = std::move(s);
        }
    }
}

void fc() {
    std::cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v, t;
        std::cin >> u >> v >> t;
        g[u].push_back(v);
        if (t == 2) g[v].push_back(u);
    }

    for (int i = 1; i <= n; i++)
        if (!dfn[i]) tarjan(i);

    std::cout << mx << "\n";
    std::sort(ans.begin(), ans.end());
    for (int x : ans) std::cout << x << " ";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
