```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

constexpr int N = 10005;

std::vector<int> g[N];
int dfn[N], low[N], timer = 0;
int scc[N], scc_cnt;
std::vector<int> stk;
bool in_stk[N];
int n, m;

int sccarr[N], sccl[N], sccr[N], idx = 0;

void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk.push_back(u);
    in_stk[u] = true;

    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
        } else if (in_stk[v]) {
            low[u] = std::min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        ++scc_cnt;
        int v;
        sccl[scc_cnt] = idx + 1;
        do {
            v = stk.back();
            stk.pop_back();
            in_stk[v] = false;
            scc[v] = scc_cnt;
            sccarr[++idx] = v;
        } while (v != u);
        sccr[scc_cnt] = idx;
    }
}

void fc() {
    std::cin >> n >> m;

    for (int i = 0; i < m; i++) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
    }

    for (int i = 1; i <= n; i++)
        if (!dfn[i]) tarjan(i);

    std::cout << scc_cnt << "\n";

    std::vector<bool> vis(scc_cnt + 1);
    for (int i = 1; i <= n; i++) {
        if (vis[scc[i]]) continue;
        vis[scc[i]] = true;
        std::sort(sccarr + sccl[scc[i]], sccarr + sccr[scc[i]] + 1);
        for (int j = sccl[scc[i]]; j <= sccr[scc[i]]; j++) std::cout << sccarr[j] << " ";
        std::cout << "\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
```
