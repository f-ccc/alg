#include <bits/stdc++.h>
using namespace std;
using ll = long long;
constexpr int N=8005;
constexpr int M=24005;

int head[N],to[M],nxt[M],tot;
void add(int u,int v){
    nxt[++tot]=head[u];
    head[u]=tot;
    to[tot]=v;
}
int dfn[N],low[N],timer;
int scc[N],scc_cnt;
std::vector<int>stk;
bool in_stk[N]; 

int n,m;
std::map<std::string,int>id;

void tarjan(int u){
    dfn[u]=low[u]=++timer;
    stk.push_back(u);
    in_stk[u]=true;

    for(int i=head[u];i;i=nxt[i]){
        int v=to[i];
        if(!dfn[v]){
            tarjan(v);
            low[u]=std::min(low[u],low[v]);
        }else if(in_stk[v]){
            low[u]=std::min(low[u],dfn[v]);
        }
    }

    if(dfn[u]==low[u]){
        ++scc_cnt;
        int v;
        do{
            v=stk.back();
            stk.pop_back();
            scc[v]=scc_cnt;
            in_stk[v]=false; 
        }while(v!=u);
    }
}
void fc() {
    std::cin>>n;
    for(int i=1;i<=n;i++){
        std::string a,b;std::cin>>a>>b;
        id[a]=i;
        id[b]=i+n;
        add(i,i+n);
    }
    std::cin>>m;
    for(int i=0;i<m;i++){
        std::string a,b;
        std::cin>>a>>b;
        add(id[b],id[a]);
    }

    for(int i=1;i<=n*2;i++)if(!dfn[i])tarjan(i);

    for(int i=1;i<=n;i++){
        std::cout<<(scc[i]!=scc[i+n]?"Safe":"Unsafe")<<"\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    //std::cin>>t;
    while (t--) fc();
    return 0;
}
