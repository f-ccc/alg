```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 3e4 + 4;
int n, q;
std::vector<int> g[N];
int dep[N], fa[N], sz[N], son[N], top[N], dfn[N], seg[N], timer;
int a[N];
int mx[N << 2], tree[N << 2];
void dfs1(int u, int f) {
    dep[u] = dep[f] + 1, fa[u] = f, son[u] = 0, sz[u] = 1;
    for (int v : g[u]) {
        if (v != f) {
            dfs1(v, u);
            sz[u] += sz[v];
            if (sz[v] > sz[son[u]]) son[u] = v;
        }
    }
}
void dfs2(int u, int tp) {
    top[u] = tp, dfn[u] = ++timer, seg[timer] = u;
    if (son[u]) dfs2(son[u], tp);
    for (int v : g[u]) {
        if (v == son[u] || v == fa[u]) continue;
        dfs2(v, v);
    }
}

inline void pushup(int p) {
    tree[p] = tree[p << 1] + tree[p << 1 | 1];
    mx[p] = std::max(mx[p << 1], mx[p << 1 | 1]);
}

void build(int p, int l, int r) {
    if (l == r) {
        mx[p] = tree[p] = a[seg[l]];
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    pushup(p);
}
void update(int p, int l, int r, int pos, int val) {
    if (l == r) {
        tree[p] = mx[p] = val;
        return;
    }
    int mid = (l + r) >> 1;
    if (pos <= mid)
        update(p << 1, l, mid, pos, val);
    else
        update(p << 1 | 1, mid + 1, r, pos, val);
    pushup(p);
}
int query_mx(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) {
        return mx[p];
    }
    int mid = (l + r) >> 1;
    int res = INT_MIN;
    if (ql <= mid) res = std::max(res, query_mx(p << 1, l, mid, ql, qr));
    if (qr > mid) res = std::max(res, query_mx(p << 1 | 1, mid + 1, r, ql, qr));
    return res;
}
int query_sum(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) {
        return tree[p];
    }
    int mid = (l + r) >> 1;
    int res = 0;
    if (ql <= mid) res += query_sum(p << 1, l, mid, ql, qr);
    if (qr > mid) res += query_sum(p << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
int path_mx(int u, int v) {
    int res = INT_MIN;
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) std::swap(u, v);
        res = std::max(res, query_mx(1, 1, n, dfn[top[u]], dfn[u]));
        u = fa[top[u]];
    }
    if (dep[u] < dep[v]) std::swap(u, v);
    return std::max(res, query_mx(1, 1, n, dfn[v], dfn[u]));
}
int path_sum(int u, int v) {
    int res = 0;
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) std::swap(u, v);
        res += query_sum(1, 1, n, dfn[top[u]], dfn[u]);
        u = fa[top[u]];
    }
    if (dep[u] < dep[v]) std::swap(u, v);
    return res + query_sum(1, 1, n, dfn[v], dfn[u]);
}
void fc() {
    std::cin >> n;
    for (int i = 1; i < n; i++) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    for (int i = 1; i <= n; i++) std::cin >> a[i];
    dfs1(1, 0);
    dfs2(1, 1);

    build(1, 1, n);

    std::cin >> q;
    while (q--) {
        std::string s;
        int u, v;
        std::cin >> s >> u >> v;
        if (s[1] == 'H') {
            update(1, 1, n, dfn[u], v);
        } else if (s[1] == 'M') {
            std::cout << path_mx(u, v) << "\n";
        } else {
            std::cout << path_sum(u, v) << "\n";
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
