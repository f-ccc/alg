#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 5;
std::vector<int> g[N];
int fa[N], dep[N], sz[N], son[N], top[N], dfn[N], seg[N], timer;
int tree[N << 2], lazy[N << 2], b[N], a[N];
int n, m, root, mod;
void modif(int& a, ll b) { a = (a + b) % mod; }
inline void pushup(int p) {
    tree[p]=(tree[p<<1]+tree[p<<1|1])%mod;
}
inline void pushdown(int p, int l, int r) {
    if (!lazy[p]) return;
    int mid = (l + r) >> 1;
    int ls = p << 1, rs = p << 1 | 1;

    modif(tree[ls], 1LL*lazy[p] * (mid - l + 1)%mod);
    modif(lazy[ls], lazy[p]);

    modif(tree[rs], 1LL*lazy[p] * (r - mid)%mod);
    modif(lazy[rs], lazy[p]);

    lazy[p] = 0;
}
void build(int p, int l, int r) {
    lazy[p] = 0;
    if (l == r) {
        tree[p] = a[l]%mod;
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    pushup(p);
}
void update(int p, int l, int r, int ql, int qr, int val) {
    if (ql <= l && qr >= r) {
        modif(tree[p], 1LL * val * (r - l + 1) % mod);
        modif(lazy[p], val);
        return;
    }
    pushdown(p, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(p << 1, l, mid, ql, qr, val);
    if (qr > mid) update(p << 1 | 1, mid + 1, r, ql, qr, val);
    pushup(p);
}
int query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tree[p];
    pushdown(p, l, r);
    int res = 0;
    int mid = (l + r) >> 1;
    if (ql <= mid) modif(res, query(p << 1, l, mid, ql, qr));
    if (mid < qr) modif(res, query(p << 1 | 1, mid + 1, r, ql, qr));
    return res;
}

void dfs1(int u, int f) {
    dep[u] = dep[f] + 1, fa[u] = f, sz[u] = 1, son[u] = 0;
    for (int v : g[u]) {
        if (v == f) continue;
        dfs1(v, u);
        sz[u] += sz[v];
        if (sz[v] > sz[son[u]]) son[u] = v;
    }
}

void dfs2(int u, int tp) {
    top[u] = tp, dfn[u] = ++timer, seg[timer] = u;
    if (son[u]) dfs2(son[u], tp);
    for (int v : g[u]) {
        if (v == fa[u] || v == son[u]) continue;
        dfs2(v, v);
    }
}

void path(int u, int v, int val) {
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) std::swap(u, v);
        update(1, 1, n, dfn[top[u]], dfn[u], val);
        u = fa[top[u]];
    }
    if (dep[u] < dep[v]) std::swap(u, v);
    update(1, 1, n, dfn[v], dfn[u], val);
}
int path_query(int u, int v) {
    int res = 0;
    while (top[u] != top[v]) {
        if (dep[top[u]] < dep[top[v]]) std::swap(u, v);
        modif(res, query(1, 1, n, dfn[top[u]], dfn[u]));
        u = fa[top[u]];
    }
    if (dep[u] < dep[v]) std::swap(u, v);
    modif(res, query(1, 1, n, dfn[v], dfn[u]));
    return res;
}
void fc() {
    std::cin >> n >> m >> root >> mod;
    for (int i = 1; i <= n; i++) std::cin >> b[i];
    for (int i = 0; i < n - 1; i++) {
        int u, v;
        std::cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs1(root, 0);
    dfs2(root, root);

    for (int i = 1; i <= n; i++) a[i] = b[seg[i]];
    build(1, 1, n);

    while (m--) {
        int op;
        std::cin >> op;
        if (op == 1) {
            int x, y, z;
            std::cin >> x >> y >> z;
            path(x, y, z);
        } else if (op == 2) {
            int x, y;
            std::cin >> x >> y;
            std::cout << path_query(x, y) << "\n";
        } else if (op == 3) {
            int x, z;
            std::cin >> x >> z;
            update(1, 1, n, dfn[x], dfn[x] + sz[x] - 1, z);
        } else {
            int x;
            std::cin >> x;
            std::cout << query(1, 1, n, dfn[x], dfn[x] + sz[x] - 1) << "\n";
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    // std::cin>>t;
    while (t--) fc();
    return 0;
}
