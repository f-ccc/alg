#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N=5e5+5;
int n,m,root;
std::vector<int>g[N];
int fa[N],dep[N],sz[N],son[N],top[N],dfn[N],seg[N],timer;
void dfs1(int u,int f){
    dep[u]=dep[f]+1,fa[u]=f,sz[u]=1,son[u]=0;
    for(int v:g[u]){
        if(v==f)continue;
        dfs1(v,u);
        sz[u]+=sz[v];
        if(sz[v]>sz[son[u]])son[u]=v;
    }
}
void dfs2(int u,int tp){
    top[u]=tp,dfn[u]=++timer,seg[timer]=u;
    if(son[u])dfs2(son[u],tp);
    for(int v:g[u]){
        if(v==fa[u]||v==son[u])continue;
        dfs2(v,v);
    }
}
int lca(int u,int v){
    while(top[u]!=top[v]){
        if(dep[top[u]]<dep[top[v]])std::swap(u,v);
        u=fa[top[u]];
    }
    return dep[u]>dep[v]?v:u;
}
void fc() {
    std::cin>>n>>m>>root;
    for(int i=0;i<n-1;i++){
        int u,v;std::cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }

    dfs1(root,0);
    dfs2(root,root);

    while(m--){
        int u,v;std::cin>>u>>v;
        std::cout<<lca(u,v)<<"\n";
    }
}

int main() {
    ios::sync_with_stdio(false);
