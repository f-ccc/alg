题目要求$z[0]=n$
```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
std::vector<int> zArray(const std::string& s){
    int n=s.size();
    std::vector<int>z(n);
    z[0]=n;
    for(int i=1,r=0,c=0;i<n;i++){
        int len=(r>i?std::min(z[i-c],r-i):0);
        while(i+len<n&&s[i+len]==s[len])++len;
        if(i+len>r)r=i+len,c=i;
        z[i]=len;
    }
    return z;
}
std::vector<int> eArray(const std::string& a,const std::string& b,const std::vector<int>& z){
    int n=a.size(), m=b.size();
    std::vector<int>e(n);
    for(int i=0,r=0,c=0;i<n;i++){
        int len=(r>i?std::min(z[i-c],r-i):0);
        while(i+len<n&&len<m&&a[i+len]==b[len])++len;
        if(i+len>r)r=i+len,c=i;
        e[i]=len;
    }
    return e;
    
}
void fc() {
    std::string a,b;std::cin>>a>>b;

    auto z=zArray(b);
    auto e=eArray(a,b,z);

    ll z_val=0,e_val=0;
    for(int i=0;i<z.size();i++)z_val^=(ll)(i+1)*(z[i]+1);
    for(int i=0;i<e.size();i++)e_val^=(ll)(i+1)*(e[i]+1);

    std::cout<<z_val<<"\n"<<e_val;
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int t = 1;
	//std::cin>>t;
	while (t--) fc();
	return 0;
}
```
