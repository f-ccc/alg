# 强连通分量

```c++
constexpr int N = 2e5 + 5;
constexpr int M = 2e5 + 5;

// 原图
int head[N], to[M], nxt[M], tot;
void add(int u, int v) {
    nxt[++tot] = head[u];
    head[u] = tot;
    to[tot] = v;
}

// Tarjan 状态
int dfn[N], low[N], timer;
int scc[N], scc_cnt;
int stk[N], top;

// 缩点后新图 (DAG)
int h2[N], to2[M], nxt2[M], tot2;
int in_deg[N];

void add2(int u, int v) {
    nxt2[++tot2] = h2[u];
    h2[u] = tot2;
    to2[tot2] = v;
    ++in_deg[v];
}

// O(V) 清空：必须在读入当前用例的 n, m 后立即调用
void clear(int n) {
    for (int i = 1; i <= n; ++i) {
        head[i] = dfn[i] = low[i] = scc[i] = h2[i] = in_deg[i] = 0;
    }
    tot = tot2 = 0;
    timer = scc_cnt = top = 0;
}

void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    stk[++top] = u;

    for (int i = head[u]; i; i = nxt[i]) {
        int v = to[i];
        if (!dfn[v]) {
            tarjan(v);
            low[u] = std::min(low[u], low[v]);
        } else if (!scc[v]) {
            low[u] = std::min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        ++scc_cnt;
        int v;
        do {
            v = stk[top--];
            scc[v] = scc_cnt;
        } while (v != u);
    }
}

// 带排序去重的 O(V + E log E) 规范重构
struct Edge {
    int u, v;
    bool operator<(const Edge& o) const {
        return u == o.u ? v < o.v : u < o.u;
    }
    bool operator==(const Edge& o) const {
        return u == o.u && v == o.v;
    }
} raw_edges[M];

void rebuild(int n) {
    int edge_cnt = 0;
    for (int u = 1; u <= n; ++u) {
        for (int i = head[u]; i; i = nxt[i]) {
            int v = to[i];
            if (scc[u] != scc[v]) {
                raw_edges[++edge_cnt] = {scc[u], scc[v]};
            }
        }
    }
    std::sort(raw_edges + 1, raw_edges + edge_cnt + 1);
    int unique_cnt = std::unique(raw_edges + 1, raw_edges + edge_cnt + 1) - (raw_edges + 1);
    for (int i = 1; i <= unique_cnt; ++i) {
        add2(raw_edges[i].u, raw_edges[i].v);
    }
}

// 规范调用入口
void solve_scc(int n) {
    for (int i = 1; i <= n; ++i) {
        if (!dfn[i]) tarjan(i);
    }
    rebuild(n);
}
```
