=== 线段树区间和
```c++
const int MAXN = 2e5 + 5; 
using T = long long; //灵活调整类型
T tree[MAXN << 2];
T a[MAXN];

inline void pushup(int p) {
    tree[p] = tree[p << 1] + tree[p << 1 | 1];
}

void build(int p, int l, int r) {
    if (l == r) {
        tree[p] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    pushup(p);
}

void update(int p, int l, int r, int pos, const T& val) {
    if (l == r) {
        tree[p] = val;
        return;
    }
    int mid = (l + r) >> 1;
    if (pos <= mid) update(p << 1, l, mid, pos, val);
    else update(p << 1 | 1, mid + 1, r, pos, val);
    pushup(p);
}

T query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tree[p];
    int mid = (l + r) >> 1;
    T res = 0;
    if (ql <= mid) res += query(p << 1, l, mid, ql, qr);
    if (qr > mid)  res += query(p << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```
=== 线段树区间和(封装)
```c++
template <typename T>
struct SegTree {
    int n;
    std::vector<T> tree;
    SegTree(const std::vector<T>& a) : n(a.size()), tree(n << 2) {
        if (n > 0) build(1, 0, n - 1, a);
    }

    void build(int node, int l, int r, const std::vector<T>& a) {
        if (l == r) {
            tree[node] = a[l];
            return;
        }
        int mid = l + ((r - l) >> 1);
        build(node << 1, l, mid, a);
        build(node << 1 | 1, mid + 1, r, a);
        tree[node] = tree[node << 1] + tree[node << 1 | 1];
    }

    void update(int node, int l, int r, int pos, const T& val) {
        if (l == r) {
            tree[node] = val;
            return;
        }
        int mid = l + ((r - l) >> 1);
        if (pos <= mid) update(node << 1, l, mid, pos, val);
        else update(node << 1 | 1, mid + 1, r, pos, val);
        tree[node] = tree[node << 1] + tree[node << 1 | 1];
    }

    T query(int node, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tree[node];
        int mid = l + ((r - l) >> 1);
        T res = T(); 
        if (ql <= mid) res += query(node << 1, l, mid, ql, qr);
        if (qr > mid)  res += query(node << 1 | 1, mid + 1, r, ql, qr);
        return res;
    }
    void update(int pos, const T& val) { update(1, 0, n - 1, pos, val); }
    T query(int ql, int qr) { return query(1, 0, n - 1, ql, qr); }
};
```
=== 线段树区间和范围修改
```c++
#include <iostream>

using namespace std;

using T = long long;
const int MAXN = 2e5 + 5;
T a[MAXN];
T tree[MAXN << 2];
T lazy[MAXN << 2];

inline void pushup(int p) {
    tree[p] = tree[p << 1] + tree[p << 1 | 1];
}

inline void pushdown(int p, int l, int r) {
    if (!lazy[p]) return;
    int mid = (l + r) >> 1;
    int ls = p << 1, rs = p << 1 | 1;

    tree[ls] += lazy[p] * (mid - l + 1);
    lazy[ls] += lazy[p];

    tree[rs] += lazy[p] * (r - mid);
    lazy[rs] += lazy[p];

    lazy[p] = 0;
}

void build(int p, int l, int r) {
    lazy[p] = 0;
    if (l == r) {
        tree[p] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    pushup(p);
}

void update(int p, int l, int r, int ql, int qr, T val) {
    if (ql <= l && r <= qr) {
        tree[p] += val * (r - l + 1);
        lazy[p] += val;
        return;
    }
    pushdown(p, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(p << 1, l, mid, ql, qr, val);
    if (qr > mid)  update(p << 1 | 1, mid + 1, r, ql, qr, val);
    pushup(p);
}

T query(int p, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tree[p];
    pushdown(p, l, r);
    int mid = (l + r) >> 1;
    T res = 0;
    if (ql <= mid) res += query(p << 1, l, mid, ql, qr);
    if (qr > mid)  res += query(p << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```

=== 线段树区间和范围修改(封装)
```c++
template <typename T>
class SegmentTreeSum {
private:
    vector<T> tree;
    vector<T> lazy;
    vector<T> data;
    int n;

    void push_down(int node, int l, int r) {
        if (lazy[node] == 0 || l == r) return;
        int mid = (l + r) >> 1;
        int left_node = node << 1;
        int right_node = node << 1 | 1;
        tree[left_node] += lazy[node] * (mid - l + 1);
        lazy[left_node] += lazy[node];
        tree[right_node] += lazy[node] * (r - mid);
        lazy[right_node] += lazy[node];
        lazy[node] = 0;
    }
    void build(int node, int l, int r) {
        if (l == r) {
            tree[node] = data[l];
            return;
        }
        int mid = (l + r) >> 1;
        build(node << 1, l, mid);
        build(node << 1 | 1, mid + 1, r);
        tree[node] = tree[node << 1] + tree[node << 1 | 1];
    }
    void update(int node, int l, int r, int pos, const T& val) {
        if (l == r) {
            tree[node] = val;
            return;
        }
        push_down(node, l, r);
        int mid = (l + r) >> 1;
        if (pos <= mid) update(node << 1, l, mid, pos, val);
        else update(node << 1 | 1, mid + 1, r, pos, val);
        tree[node] = tree[node << 1] + tree[node << 1 | 1];
    }

    void range_update(int node, int l, int r, int ul, int ur, const T& add_val) {
        if (ul <= l && r <= ur) {
            tree[node] += add_val * (r - l + 1);
            lazy[node] += add_val;
            return;
        }
        push_down(node, l, r);
        int mid = (l + r) >> 1;
        if (ul <= mid) range_update(node << 1, l, mid, ul, ur, add_val);
        if (ur > mid) range_update(node << 1 | 1, mid + 1, r, ul, ur, add_val);
        tree[node] = tree[node << 1] + tree[node << 1 | 1];
    }

    T query(int node, int l, int r, int ql, int qr) {
        if (qr < l || ql > r) return T();
        if (ql <= l && r <= qr) return tree[node];
        push_down(node, l, r);
        int mid = (l + r) >> 1;
        return query(node << 1, l, mid, ql, qr) + query(node << 1 | 1, mid + 1, r, ql, qr);
    }
public:
    SegmentTreeSum(const vector<T>& arr) : data(arr), n(arr.size()) {
        tree.resize(n << 2);
        lazy.resize(n << 2, 0);
        build(1, 0, n - 1);
    }
    void update(int pos, const T& val) {
        if (pos < 0 || pos >= n) {
            cerr << "pos out of range!" << endl;
            return;
        }
        update(1, 0, n - 1, pos, val);
    }
    void rangeUpdate(int ul, int ur, const T& add_val) {
        if (ul < 0 || ur >= n || ul > ur) {
            cerr << "update range out of range!" << endl;
            return;
        }
        range_update(1, 0, n - 1, ul, ur, add_val);
    }
    T query(int ql, int qr) {
        if (ql < 0 || qr >= n || ql > qr) {
            cerr << "query range out of range!" << endl;
            return T();
        }
        return query(1, 0, n - 1, ql, qr);
    }
};
```
