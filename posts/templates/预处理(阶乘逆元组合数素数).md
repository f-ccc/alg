# 预处理
```c++
using ll = long long;
const int N = 1e6 + 5;
const int MOD = 1e9 + 7;

// ==================== 1. 阶乘与组合数预处理 ====================
ll fact[N], inv[N];

ll qpow(ll a, ll b) {
    ll res = 1;
    for (a %= MOD; b > 0; b >>= 1, a = a * a % MOD) {
        if (b & 1) res = res * a % MOD;
    }
    return res;
}

void init_fac(int n) {
    fact[0] = 1;
    for (int i = 1; i <= n; ++i) fact[i] = fact[i - 1] * i % MOD;
    inv[n] = qpow(fact[n], MOD - 2);
    for (int i = n - 1; i >= 0; --i) inv[i] = inv[i + 1] * (i + 1) % MOD;
}

inline ll C(int n, int k) {
    if (n < 0 || k < 0 || k > n) return 0;
    return fact[n] * inv[k] % MOD * inv[n - k] % MOD;
}

// ==================== 2. 静态线性筛（欧拉筛） ====================
int primes[N], p_cnt;
bool vis[N]; // vis[i] 为 true 表示 i 为合数

void init_prime(int n) {
    vis[0] = vis[1] = true;
    p_cnt = 0;
    for (int i = 2; i <= n; ++i) {
        if (!vis[i]) primes[p_cnt++] = i;
        for (int j = 0; j < p_cnt && primes[j] <= n / i; ++j) {
            vis[i * primes[j]] = true;
            if (i % primes[j] == 0) break; // 保证每个合数只被其最小质因数筛去
        }
    }
}
```
