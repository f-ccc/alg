# 割点和点双连通分量

:::tabs variant:code
=== 割点
```c++
#include <bits/stdc++.h>
using namespace std;

const int N = 20005;
vector<int> g[N];
int dfn[N], low[N], tot;
bool cut[N];

void tarjan(int u, bool root) {
    dfn[u] = low[u] = ++tot;
    int son = 0;
    for (int v : g[u]) {
        if (!dfn[v]) {
            son++;
            tarjan(v, false);
            low[u] = min(low[u], low[v]); 
            if (low[v] >= dfn[u] && (!root || son >= 2)) cut[u] = true;
        } else {
            low[u] = min(low[u], dfn[v]); // 弃边 dfn[v] > dfn[u] 不影响最小值
        }
    } 
}
```
=== 点双连通分量
```c++
#include <bits/stdc++.h>
using namespace std;

const int N = 500005;
vector<int> g[N], bcc[N];
int dfn[N], low[N], stk[N], tot, top, cnt;

void tarjan(int u, bool root) {
    dfn[u] = low[u] = ++tot;
    stk[++top] = u;
    if (root && g[u].empty()) {
        bcc[++cnt].push_back(u);
        return;
    }
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v, false);
            low[u] = min(low[u], low[v]);
            if (low[v] >= dfn[u]) {
                cnt++;
                int x;
                do {
                    x = stk[top--];
                    bcc[cnt].push_back(x);
                } while (x != v);
                bcc[cnt].push_back(u);
            }
        } else {
            low[u] = min(low[u], dfn[v]);
        }
    }
}
```
